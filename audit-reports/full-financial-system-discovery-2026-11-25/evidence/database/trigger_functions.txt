audit_employee_changes|CREATE OR REPLACE FUNCTION public.audit_employee_changes()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
    v_action VARCHAR(20);
    v_old_values JSONB;
    v_new_values JSONB;
    v_entity_name VARCHAR(200);
BEGIN
    -- Determine action
    IF TG_OP = 'INSERT' THEN
        v_action := 'create';
        v_old_values := NULL;
        v_new_values := to_jsonb(NEW);
        v_entity_name := NEW.first_name || ' ' || NEW.last_name;
    ELSIF TG_OP = 'UPDATE' THEN
        v_action := 'update';
        v_old_values := to_jsonb(OLD);
        v_new_values := to_jsonb(NEW);
        v_entity_name := NEW.first_name || ' ' || NEW.last_name;
    ELSIF TG_OP = 'DELETE' THEN
        v_action := 'delete';
        v_old_values := to_jsonb(OLD);
        v_new_values := NULL;
        v_entity_name := OLD.first_name || ' ' || OLD.last_name;
    END IF;

    -- Create audit log (use COALESCE to handle cases where user context is not set)
    PERFORM create_hr_audit_log(
        COALESCE(NEW.tenant_id, OLD.tenant_id),
        COALESCE(NEW.restaurant_id, OLD.restaurant_id),
        'employee',
        COALESCE(NEW.id, OLD.id),
        v_entity_name,
        v_action,
        v_old_values,
        v_new_values,
        COALESCE(NEW.updated_by, NEW.created_by, OLD.updated_by),
        NULL
    );

    IF TG_OP = 'DELETE' THEN
        RETURN OLD;
    ELSE
        RETURN NEW;
    END IF;
END;
$function$

refresh_invoice_supplier_totals|CREATE OR REPLACE FUNCTION public.refresh_invoice_supplier_totals()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
  target_invoice_id INT;
BEGIN
  target_invoice_id := COALESCE(NEW.invoice_id, OLD.invoice_id);

  UPDATE invoices_supplier inv
  SET subtotal = COALESCE(calc.subtotal, 0),
      tax_amount = COALESCE(calc.tax_amount, 0),
      discount_amount = COALESCE(calc.discount_amount, 0),
      total_amount = COALESCE(calc.total_amount, 0),
      balance_due = GREATEST(COALESCE(calc.total_amount, 0) - COALESCE(inv.amount_paid, 0), 0),
      updated_at = CURRENT_TIMESTAMP
  FROM (
    SELECT
      invoice_id,
      ROUND(COALESCE(SUM(quantity * unit_price), 0), 2) AS subtotal,
      ROUND(COALESCE(SUM(tax_amount), 0), 2) AS tax_amount,
      ROUND(COALESCE(SUM(discount_amount), 0), 2) AS discount_amount,
      ROUND(COALESCE(SUM(line_total), 0), 2) AS total_amount
    FROM invoice_supplier_lines
    WHERE invoice_id = target_invoice_id
    GROUP BY invoice_id
  ) calc
  WHERE inv.id = target_invoice_id;

  UPDATE invoices_supplier
  SET subtotal = 0,
      tax_amount = 0,
      discount_amount = 0,
      total_amount = 0,
      balance_due = 0,
      updated_at = CURRENT_TIMESTAMP
  WHERE id = target_invoice_id
    AND NOT EXISTS (
      SELECT 1 FROM invoice_supplier_lines WHERE invoice_id = target_invoice_id
    );

  RETURN COALESCE(NEW, OLD);
END;
$function$

calculate_salary_totals|CREATE OR REPLACE FUNCTION public.calculate_salary_totals()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
    -- Calculate gross salary (base + all earnings)
    NEW.gross_salary := COALESCE(NEW.base_salary, 0) +
                        COALESCE(NEW.overtime_amount, 0) +
                        COALESCE(NEW.bonus, 0) +
                        COALESCE(NEW.commission, 0) +
                        COALESCE(NEW.allowances, 0) +
                        COALESCE(NEW.tips, 0) +
                        COALESCE(NEW.other_earnings, 0);

    -- Calculate total deductions
    NEW.total_deductions := COALESCE(NEW.tax, 0) +
                           COALESCE(NEW.social_insurance, 0) +
                           COALESCE(NEW.health_insurance, 0) +
                           COALESCE(NEW.pension, 0) +
                           COALESCE(NEW.loan_deduction, 0) +
                           COALESCE(NEW.advance_deduction, 0) +
                           COALESCE(NEW.other_deductions, 0);

    -- Calculate net salary
    NEW.net_salary := NEW.gross_salary - NEW.total_deductions;

    -- Calculate overtime amount if not set
    IF NEW.overtime_hours > 0 AND NEW.overtime_rate > 0 THEN
        NEW.overtime_amount := NEW.overtime_hours * NEW.overtime_rate;
    END IF;

    RETURN NEW;
END;

calculate_leave_days|CREATE OR REPLACE FUNCTION public.calculate_leave_days()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
    -- Calculate total days if not set
    IF NEW.is_half_day THEN
        NEW.total_days := 0.5;
    ELSE
        -- Calculate business days (excluding weekends)
        NEW.total_days := (
            SELECT COUNT(*)
            FROM generate_series(NEW.start_date, NEW.end_date, '1 day'::interval) AS d

cfr002_guard_journal_header_write|CREATE OR REPLACE FUNCTION public.cfr002_guard_journal_header_write()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'pg_catalog', 'public'
AS $function$
DECLARE
    restaurant_tenant_id INT;
BEGIN
    IF TG_OP = 'INSERT' THEN
        IF NEW.status <> 'draft' THEN
            RAISE EXCEPTION 'new journal entries must start as draft' USING ERRCODE = '23514';
        END IF;
        SELECT tenant_id INTO restaurant_tenant_id
        FROM public.restaurants WHERE id = NEW.restaurant_id FOR UPDATE;
        IF restaurant_tenant_id IS NULL OR restaurant_tenant_id <> NEW.tenant_id THEN
            RAISE EXCEPTION 'journal restaurant does not belong to its tenant' USING ERRCODE = '23514';
        END IF;
        RETURN NEW;
    END IF;

    IF TG_OP = 'DELETE' THEN
        IF OLD.status <> 'draft' THEN
            RAISE EXCEPTION 'posted or void journals cannot be deleted' USING ERRCODE = '23514';
        END IF;
        RETURN OLD;
    END IF;

    IF OLD.status = 'draft' THEN
        IF NEW.status NOT IN ('draft', 'posted', 'void') THEN
            RAISE EXCEPTION 'invalid journal status transition from draft to %', NEW.status USING ERRCODE = '23514';
        END IF;
        IF NEW.restaurant_id <> OLD.restaurant_id OR NEW.tenant_id <> OLD.tenant_id THEN
            SELECT tenant_id INTO restaurant_tenant_id
            FROM public.restaurants WHERE id = NEW.restaurant_id FOR UPDATE;
            IF restaurant_tenant_id IS NULL OR restaurant_tenant_id <> NEW.tenant_id THEN
                RAISE EXCEPTION 'journal restaurant does not belong to its tenant' USING ERRCODE = '23514';
            END IF;
        END IF;
        IF NEW.status = 'posted' AND (NEW.posted_at IS NULL OR NEW.posted_by IS NULL) THEN
            RAISE EXCEPTION 'posting requires posted_at and posted_by audit fields' USING ERRCODE = '23514';
        END IF;
        IF NEW.status = 'void' AND (NEW.voided_at IS NULL OR NEW.voided_by IS NULL OR btrim(COALESCE(NEW.void_reason, '')) = '') THEN
            RAISE EXCEPTION 'voiding a draft requires void audit fields and a reason' USING ERRCODE = '23514';
        END IF;
        RETURN NEW;
    END IF;

    IF OLD.status = 'posted' AND NEW.status = 'void' THEN
        IF ROW(NEW.id, NEW.tenant_id, NEW.restaurant_id, NEW.entry_number, NEW.entry_date,
               NEW.fiscal_period, NEW.description, NEW.description_ar, NEW.reference_type,
               NEW.reference_id, NEW.reference_number, NEW.entry_type, NEW.total_debit,
               NEW.total_credit, NEW.posted_at, NEW.posted_by, NEW.is_approved,
               NEW.approved_by, NEW.approved_at, NEW.notes, NEW.attachments,
               NEW.created_at, NEW.created_by)
           IS DISTINCT FROM
           ROW(OLD.id, OLD.tenant_id, OLD.restaurant_id, OLD.entry_number, OLD.entry_date,
               OLD.fiscal_period, OLD.description, OLD.description_ar, OLD.reference_type,
               OLD.reference_id, OLD.reference_number, OLD.entry_type, OLD.total_debit,
               OLD.total_credit, OLD.posted_at, OLD.posted_by, OLD.is_approved,
               OLD.approved_by, OLD.approved_at, OLD.notes, OLD.attachments,
               OLD.created_at, OLD.created_by) THEN
            RAISE EXCEPTION 'voiding cannot modify posted journal accounting history' USING ERRCODE = '23514';
        END IF;
        IF NEW.voided_at IS NULL OR NEW.voided_by IS NULL OR btrim(COALESCE(NEW.void_reason, '')) = '' THEN
            RAISE EXCEPTION 'voiding a posted journal requires void audit fields and a reason' USING ERRCODE = '23514';
        END IF;
        RETURN NEW;
    END IF;

    RAISE EXCEPTION 'journal status % is immutable; permitted transitions are draft to posted/void and posted to audited void', OLD.status
        USING ERRCODE = '23514';
END;

calculate_attendance_hours|CREATE OR REPLACE FUNCTION public.calculate_attendance_hours()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
    work_minutes INTEGER;
BEGIN
    -- Calculate total hours if both clock_in and clock_out are present
    IF NEW.clock_in IS NOT NULL AND NEW.clock_out IS NOT NULL THEN
        -- Calculate work minutes (total time minus breaks)
        work_minutes := EXTRACT(EPOCH FROM (NEW.clock_out - NEW.clock_in)) / 60 - COALESCE(NEW.total_break_minutes, 0);
        NEW.total_hours := ROUND((work_minutes / 60.0)::numeric, 2);

        -- Calculate overtime (assuming 8 hours is regular)
        IF NEW.total_hours > 8 THEN
            NEW.regular_hours := 8;
            NEW.overtime_hours := ROUND((NEW.total_hours - 8)::numeric, 2);
            NEW.is_overtime := true;
        ELSE
            NEW.regular_hours := NEW.total_hours;
            NEW.overtime_hours := 0;
            NEW.is_overtime := false;
        END IF;

        -- Check if late
        IF NEW.scheduled_clock_in IS NOT NULL THEN
            IF EXTRACT(EPOCH FROM (NEW.clock_in::time - NEW.scheduled_clock_in)) > 0 THEN
                NEW.is_late := true;
                NEW.late_by_minutes := EXTRACT(EPOCH FROM (NEW.clock_in::time - NEW.scheduled_clock_in)) / 60;
            END IF;
        END IF;

        -- Check if early departure
        IF NEW.scheduled_clock_out IS NOT NULL THEN
            IF EXTRACT(EPOCH FROM (NEW.scheduled_clock_out - NEW.clock_out::time)) > 0 THEN
                NEW.is_early_departure := true;
                NEW.early_departure_minutes := EXTRACT(EPOCH FROM (NEW.scheduled_clock_out - NEW.clock_out::time)) / 60;
            END IF;
        END IF;
    END IF;

    RETURN NEW;
END;
$function$

 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'pg_catalog', 'public'
AS $function$
DECLARE
    target_entry_id INT;
    journal_status TEXT;
    journal_tenant_id INT;

maintain_order_stock|CREATE OR REPLACE FUNCTION public.maintain_order_stock()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'pg_catalog', 'public'
AS $function$
DECLARE item RECORD;
BEGIN
    IF TG_OP = 'DELETE' THEN
        IF EXISTS (SELECT 1 FROM public.order_stock_reservations
                   WHERE order_id = OLD.id AND tenant_id = OLD.tenant_id AND restaurant_id = OLD.restaurant_id) THEN
            RAISE EXCEPTION 'Orders with inventory history cannot be deleted' USING ERRCODE = '23514';
        END IF;
        RETURN OLD;
    END IF;
    IF OLD.status = 'delivered' AND NEW.status = 'cancelled' THEN
        RAISE EXCEPTION 'Delivered stock requires an explicit physical return adjustment' USING ERRCODE = '23514';
    END IF;
    IF NEW.tenant_id <> OLD.tenant_id OR NEW.restaurant_id <> OLD.restaurant_id THEN
        RAISE EXCEPTION 'Order scope is immutable' USING ERRCODE = '23514';
    END IF;
    IF OLD.status = 'pending' AND NEW.status = 'confirmed' THEN
        -- Consistent lock order serializes competing orders without overselling.
        FOR item IN
            SELECT p.id, p.quantity_in_stock, q.quantity
            FROM public.products p
            JOIN (SELECT product_id, SUM(quantity - cancelled_quantity)::BIGINT AS quantity
                  FROM public.order_items
                  WHERE order_id = OLD.id AND tenant_id = OLD.tenant_id AND restaurant_id = OLD.restaurant_id
                  GROUP BY product_id HAVING SUM(quantity - cancelled_quantity) > 0) q ON q.product_id = p.id
            WHERE p.tenant_id = OLD.tenant_id AND p.restaurant_id = OLD.restaurant_id AND p.track_inventory
            ORDER BY p.id FOR UPDATE OF p
        LOOP
            IF item.quantity_in_stock IS NULL OR item.quantity <= 0 OR item.quantity_in_stock < item.quantity THEN
                RAISE EXCEPTION 'Insufficient inventory for product %', item.id USING ERRCODE = '23514';
            END IF;
            INSERT INTO public.order_stock_reservations(order_id, tenant_id, restaurant_id, product_id, quantity)
            VALUES (OLD.id, OLD.tenant_id, OLD.restaurant_id, item.id, item.quantity);
            UPDATE public.products SET quantity_in_stock = quantity_in_stock - item.quantity, updated_at = CURRENT_TIMESTAMP
            WHERE id = item.id AND tenant_id = OLD.tenant_id AND restaurant_id = OLD.restaurant_id;
            INSERT INTO public.inventory_movements(tenant_id,restaurant_id,product_id,order_id,reference_type,movement_type,quantity,notes)
            VALUES(OLD.tenant_id,OLD.restaurant_id,item.id,OLD.id,'order_stock','reservation',-item.quantity,'Stock reserved at order confirmation');
        END LOOP;
    ELSIF NEW.status = 'cancelled' AND OLD.status <> 'cancelled' THEN
        FOR item IN
            SELECT p.id, s.quantity - s.released_quantity AS quantity FROM public.order_stock_reservations s
            JOIN public.products p ON p.id = s.product_id AND p.tenant_id = s.tenant_id AND p.restaurant_id = s.restaurant_id
            WHERE s.order_id = OLD.id AND s.tenant_id = OLD.tenant_id AND s.restaurant_id = OLD.restaurant_id AND NOT s.released
            ORDER BY p.id FOR UPDATE OF p
        LOOP
            UPDATE public.products SET quantity_in_stock = quantity_in_stock + item.quantity, updated_at = CURRENT_TIMESTAMP
            WHERE id = item.id AND tenant_id = OLD.tenant_id AND restaurant_id = OLD.restaurant_id;
            INSERT INTO public.inventory_movements(tenant_id,restaurant_id,product_id,order_id,reference_type,movement_type,quantity,notes)
            VALUES(OLD.tenant_id,OLD.restaurant_id,item.id,OLD.id,'order_stock','reservation_release',item.quantity,'Unfulfilled order cancellation');
            UPDATE public.order_stock_reservations SET released = true, released_quantity = quantity, released_at = CURRENT_TIMESTAMP
            WHERE order_id = OLD.id AND product_id = item.id AND tenant_id = OLD.tenant_id AND restaurant_id = OLD.restaurant_id;
        END LOOP;
    END IF;
    RETURN NEW;
END;
$function$

prevent_overlapping_leaves|CREATE OR REPLACE FUNCTION public.prevent_overlapping_leaves()

 LANGUAGE plpgsql

BEGIN
    -- Check for overlapping approved/pending leaves
    IF EXISTS (
        SELECT 1 FROM leaves
        WHERE employee_id = NEW.employee_id
        AND id != COALESCE(NEW.id, 0)
        AND status IN ('pending', 'approved')
        AND (
            (NEW.start_date BETWEEN start_date AND end_date) OR

    END IF;

    RETURN NEW;

cfr002_assert_journal_integrity_deferred|CREATE OR REPLACE FUNCTION public.cfr002_assert_journal_integrity_deferred()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'pg_catalog', 'public'
AS $function$
DECLARE
    target_entry_id INT;
    journal_status TEXT;
    journal_posted_at TIMESTAMP;
    header_debit NUMERIC(15,2);
    header_credit NUMERIC(15,2);
    line_count BIGINT;
    line_debit NUMERIC;
    line_credit NUMERIC;
    invalid_line_id INT;
    ledger_effect_count BIGINT;
    matching_effect_count BIGINT;
BEGIN
    IF TG_TABLE_NAME = 'journal_entries' THEN
        IF TG_OP = 'DELETE' THEN target_entry_id := OLD.id; ELSE target_entry_id := NEW.id; END IF;
    ELSIF TG_TABLE_NAME = 'journal_entry_lines' THEN
        IF TG_OP = 'DELETE' THEN target_entry_id := OLD.journal_entry_id; ELSE target_entry_id := NEW.journal_entry_id; END IF;
    ELSE
        IF TG_OP = 'DELETE' THEN target_entry_id := OLD.journal_entry_id; ELSE target_entry_id := NEW.journal_entry_id; END IF;
    END IF;

    SELECT status, posted_at, total_debit, total_credit
    INTO journal_status, journal_posted_at, header_debit, header_credit
    FROM public.journal_entries WHERE id = target_entry_id;
    IF NOT FOUND THEN RETURN NULL; END IF;

    SELECT COUNT(*), COALESCE(SUM(debit_amount), 0), COALESCE(SUM(credit_amount), 0)
    INTO line_count, line_debit, line_credit
    FROM public.journal_entry_lines WHERE journal_entry_id = target_entry_id;

    IF EXISTS (
        SELECT 1
        FROM public.journal_entry_lines jel
        LEFT JOIN public.chart_of_accounts coa ON coa.id = jel.account_id
        WHERE jel.journal_entry_id = target_entry_id
          AND (jel.tenant_id <> (SELECT tenant_id FROM public.journal_entries WHERE id = target_entry_id)
               OR coa.id IS NULL
               OR coa.tenant_id <> (SELECT tenant_id FROM public.journal_entries WHERE id = target_entry_id)
               OR coa.restaurant_id <> (SELECT restaurant_id FROM public.journal_entries WHERE id = target_entry_id))
    ) THEN
        RAISE EXCEPTION 'journal % contains a line outside its tenant or restaurant scope', target_entry_id
            USING ERRCODE = '23514';
    END IF;

    IF journal_status = 'posted' THEN
        IF line_count < 2 OR line_debit <> line_credit OR
           line_debit <> header_debit OR line_credit <> header_credit THEN
            RAISE EXCEPTION 'posted journal % must contain at least two exactly balanced lines matching its header', target_entry_id
                USING ERRCODE = '23514';
        END IF;
        SELECT jel.id, COUNT(gl.id),
               COUNT(gl.id) FILTER (WHERE gl.debit_amount = jel.debit_amount AND gl.credit_amount = jel.credit_amount)
        INTO invalid_line_id, ledger_effect_count, matching_effect_count
            FROM public.journal_entry_lines jel
            LEFT JOIN public.general_ledger gl
              ON gl.journal_entry_id = target_entry_id AND gl.journal_line_id = jel.id
            WHERE jel.journal_entry_id = target_entry_id
            GROUP BY jel.id, jel.debit_amount, jel.credit_amount
            HAVING COUNT(gl.id) <> 1
                OR COUNT(gl.id) FILTER (WHERE gl.debit_amount = jel.debit_amount AND gl.credit_amount = jel.credit_amount) <> 1
            ORDER BY jel.id
            LIMIT 1;
        IF invalid_line_id IS NOT NULL THEN
            RAISE EXCEPTION 'posted journal % line % must have exactly one matching General Ledger row (found %, matching %)',
                target_entry_id, invalid_line_id, ledger_effect_count, matching_effect_count
                USING ERRCODE = '23514';
        END IF;
    ELSIF journal_status = 'void' AND journal_posted_at IS NOT NULL THEN
        IF line_count < 2 OR line_debit <> line_credit OR
           line_debit <> header_debit OR line_credit <> header_credit THEN
            RAISE EXCEPTION 'voided posted journal % must preserve its original balanced lines and header totals', target_entry_id
                USING ERRCODE = '23514';
        END IF;
        SELECT jel.id, COUNT(gl.id),
               COUNT(gl.id) FILTER (WHERE gl.debit_amount = jel.debit_amount AND gl.credit_amount = jel.credit_amount)
        INTO invalid_line_id, ledger_effect_count, matching_effect_count
            FROM public.journal_entry_lines jel
            LEFT JOIN public.general_ledger gl
              ON gl.journal_entry_id = target_entry_id AND gl.journal_line_id = jel.id
            WHERE jel.journal_entry_id = target_entry_id
            GROUP BY jel.id, jel.debit_amount, jel.credit_amount
            HAVING COUNT(gl.id) <> 2
                OR COUNT(gl.id) FILTER (WHERE gl.debit_amount = jel.debit_amount AND gl.credit_amount = jel.credit_amount) <> 1
                OR COUNT(gl.id) FILTER (WHERE gl.debit_amount = jel.credit_amount AND gl.credit_amount = jel.debit_amount) <> 1
            ORDER BY jel.id
            LIMIT 1;
        IF invalid_line_id IS NOT NULL THEN
            RAISE EXCEPTION 'voided journal % line % must retain one original and one inverse General Ledger row (found %, matching original %)',
                target_entry_id, invalid_line_id, ledger_effect_count, matching_effect_count
                USING ERRCODE = '23514';
        END IF;
    ELSE
        IF EXISTS (SELECT 1 FROM public.general_ledger WHERE journal_entry_id = target_entry_id) THEN
            RAISE EXCEPTION 'draft or unposted void journal % cannot have General Ledger effects', target_entry_id
                USING ERRCODE = '23514';
        END IF;
    END IF;
    RETURN NULL;
END;

calculate_overall_rating|CREATE OR REPLACE FUNCTION public.calculate_overall_rating()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
    rating_count INTEGER := 0;
    rating_sum DECIMAL := 0;
BEGIN
    -- Calculate average of all non-null category ratings
    IF NEW.quality_of_work IS NOT NULL THEN
        rating_sum := rating_sum + NEW.quality_of_work;
        rating_count := rating_count + 1;
    END IF;
    IF NEW.productivity IS NOT NULL THEN
        rating_sum := rating_sum + NEW.productivity;
        rating_count := rating_count + 1;
    END IF;
    IF NEW.communication IS NOT NULL THEN
        rating_sum := rating_sum + NEW.communication;
        rating_count := rating_count + 1;
    END IF;
    IF NEW.teamwork IS NOT NULL THEN
        rating_sum := rating_sum + NEW.teamwork;
        rating_count := rating_count + 1;
    END IF;
    IF NEW.punctuality IS NOT NULL THEN
        rating_sum := rating_sum + NEW.punctuality;
        rating_count := rating_count + 1;
    END IF;
    IF NEW.attendance IS NOT NULL THEN
        rating_sum := rating_sum + NEW.attendance;
        rating_count := rating_count + 1;
    END IF;
    IF NEW.initiative IS NOT NULL THEN
        rating_sum := rating_sum + NEW.initiative;
        rating_count := rating_count + 1;
    END IF;
    IF NEW.problem_solving IS NOT NULL THEN
        rating_sum := rating_sum + NEW.problem_solving;
        rating_count := rating_count + 1;
    END IF;
    IF NEW.customer_service IS NOT NULL THEN
        rating_sum := rating_sum + NEW.customer_service;
        rating_count := rating_count + 1;
    END IF;
    IF NEW.technical_skills IS NOT NULL THEN
        rating_sum := rating_sum + NEW.technical_skills;
        rating_count := rating_count + 1;
    END IF;
    IF NEW.leadership IS NOT NULL THEN
        rating_sum := rating_sum + NEW.leadership;
        rating_count := rating_count + 1;
    END IF;
    IF NEW.adaptability IS NOT NULL THEN
        rating_sum := rating_sum + NEW.adaptability;
        rating_count := rating_count + 1;
    END IF;

refresh_invoice_customer_totals|CREATE OR REPLACE FUNCTION public.refresh_invoice_customer_totals()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
  target_invoice_id INT;
BEGIN
  target_invoice_id := COALESCE(NEW.invoice_id, OLD.invoice_id);

  UPDATE invoices_customer ic
  SET subtotal = COALESCE(calc.subtotal, 0),
      tax_amount = COALESCE(calc.tax_amount, 0),
      discount_amount = COALESCE(calc.discount_amount, 0),
      total_amount = COALESCE(calc.total_amount, 0),
      balance_due = GREATEST(COALESCE(calc.total_amount, 0) - COALESCE(ic.amount_paid, 0), 0),
      updated_at = CURRENT_TIMESTAMP
  FROM (
    SELECT
      invoice_id,
      ROUND(COALESCE(SUM(quantity * unit_price), 0), 2) AS subtotal,
      ROUND(COALESCE(SUM(tax_amount), 0), 2) AS tax_amount,
      ROUND(COALESCE(SUM(discount_amount), 0), 2) AS discount_amount,
      ROUND(COALESCE(SUM(line_total), 0), 2) AS total_amount
    FROM invoice_customer_lines
    WHERE invoice_id = target_invoice_id
    GROUP BY invoice_id
  ) calc
  WHERE ic.id = target_invoice_id;

  UPDATE invoices_customer
  SET subtotal = 0,
      tax_amount = 0,
      discount_amount = 0,
      total_amount = 0,
      balance_due = 0,
      updated_at = CURRENT_TIMESTAMP
  WHERE id = target_invoice_id
    AND NOT EXISTS (
      SELECT 1 FROM invoice_customer_lines WHERE invoice_id = target_invoice_id
    );

  RETURN COALESCE(NEW, OLD);
END;
$function$

audit_role_changes|CREATE OR REPLACE FUNCTION public.audit_role_changes()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
    v_action VARCHAR(20);
    v_old_values JSONB;
    v_new_values JSONB;
BEGIN
    IF TG_OP = 'INSERT' THEN
        v_action := 'create';
        v_old_values := NULL;
        v_new_values := to_jsonb(NEW);
    ELSIF TG_OP = 'UPDATE' THEN
        v_action := 'update';
        v_old_values := to_jsonb(OLD);
        v_new_values := to_jsonb(NEW);
    ELSIF TG_OP = 'DELETE' THEN
        v_action := 'delete';
        v_old_values := to_jsonb(OLD);
        v_new_values := NULL;
    END IF;

    PERFORM create_hr_audit_log(
        COALESCE(NEW.tenant_id, OLD.tenant_id),
        COALESCE(NEW.restaurant_id, OLD.restaurant_id),
        'role',
        COALESCE(NEW.id, OLD.id),
        COALESCE(NEW.role_name, OLD.role_name),
        v_action,
        v_old_values,
        v_new_values,
        COALESCE(NEW.updated_by, NEW.created_by, OLD.updated_by),
        NULL
    );

    IF TG_OP = 'DELETE' THEN
        RETURN OLD;
    ELSE
        RETURN NEW;
    END IF;
END;

recalculate_invoice_customer_line_total|CREATE OR REPLACE FUNCTION public.recalculate_invoice_customer_line_total()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
  NEW.tax_amount := ROUND((COALESCE(NEW.quantity, 0) * COALESCE(NEW.unit_price, 0)) * (COALESCE(NEW.tax_rate, 0) / 100.0), 2);
  NEW.discount_amount := ROUND((COALESCE(NEW.quantity, 0) * COALESCE(NEW.unit_price, 0)) * (COALESCE(NEW.discount_rate, 0) / 100.0), 2);
  NEW.line_total := ROUND((COALESCE(NEW.quantity, 0) * COALESCE(NEW.unit_price, 0)) - COALESCE(NEW.discount_amount, 0) + COALESCE(NEW.tax_amount, 0), 2);
  RETURN NEW;
END;

cfr002_guard_general_ledger_insert|CREATE OR REPLACE FUNCTION public.cfr002_guard_general_ledger_insert()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'pg_catalog', 'public'
AS $function$
DECLARE
    journal_status TEXT;
    journal_tenant_id INT;
    journal_restaurant_id INT;
    journal_entry_date DATE;
    journal_fiscal_period VARCHAR(7);
    journal_voided_at TIMESTAMP;
    journal_void_reason TEXT;
    line_tenant_id INT;
    line_account_id INT;
    line_debit NUMERIC(15,2);
    line_credit NUMERIC(15,2);
    account_tenant_id INT;
    account_restaurant_id INT;
    existing_count INT;
    original_count INT;
    reverse_count INT;
BEGIN
    SELECT status, tenant_id, restaurant_id, entry_date, fiscal_period, voided_at, void_reason
    INTO journal_status, journal_tenant_id, journal_restaurant_id, journal_entry_date,
         journal_fiscal_period, journal_voided_at, journal_void_reason
    FROM public.journal_entries
    WHERE id = NEW.journal_entry_id
    FOR UPDATE;
    IF NOT FOUND OR journal_status NOT IN ('posted', 'void') THEN
        RAISE EXCEPTION 'General Ledger rows require a posted journal or its audited void transition' USING ERRCODE = '23514';
    END IF;
    IF NEW.tenant_id <> journal_tenant_id OR NEW.restaurant_id <> journal_restaurant_id THEN
        RAISE EXCEPTION 'General Ledger tenant or restaurant does not match its journal' USING ERRCODE = '23514';
    END IF;

    SELECT tenant_id, account_id, debit_amount, credit_amount
    INTO line_tenant_id, line_account_id, line_debit, line_credit
    FROM public.journal_entry_lines
    WHERE id = NEW.journal_line_id AND journal_entry_id = NEW.journal_entry_id;
    IF NOT FOUND OR NEW.tenant_id <> line_tenant_id OR NEW.account_id <> line_account_id THEN
        RAISE EXCEPTION 'General Ledger row does not match its journal line' USING ERRCODE = '23514';
    END IF;

    SELECT tenant_id, restaurant_id INTO account_tenant_id, account_restaurant_id
    FROM public.chart_of_accounts WHERE id = NEW.account_id;
    IF NOT FOUND OR account_tenant_id <> NEW.tenant_id OR account_restaurant_id <> NEW.restaurant_id THEN
        RAISE EXCEPTION 'General Ledger account is outside its tenant and restaurant' USING ERRCODE = '23514';
    END IF;

    SELECT COUNT(*),
           COUNT(*) FILTER (WHERE debit_amount = line_debit AND credit_amount = line_credit),
           COUNT(*) FILTER (WHERE debit_amount = line_credit AND credit_amount = line_debit)
    INTO existing_count, original_count, reverse_count
    FROM public.general_ledger
    WHERE journal_entry_id = NEW.journal_entry_id AND journal_line_id = NEW.journal_line_id;

    IF existing_count = 0 AND journal_status = 'posted' THEN
        IF NEW.debit_amount <> line_debit OR NEW.credit_amount <> line_credit THEN
            RAISE EXCEPTION 'initial General Ledger effect must match its journal line' USING ERRCODE = '23514';
        END IF;
        IF NEW.entry_date <> journal_entry_date OR
           NEW.fiscal_period <> COALESCE(journal_fiscal_period, to_char(journal_entry_date, 'YYYY-MM')) THEN
            RAISE EXCEPTION 'initial General Ledger period must match its journal' USING ERRCODE = '23514';
        END IF;
    ELSIF journal_status = 'void' AND journal_voided_at IS NOT NULL
          AND btrim(COALESCE(journal_void_reason, '')) <> ''
          AND existing_count = 1 AND original_count = 1 AND
          NEW.debit_amount = line_credit AND NEW.credit_amount = line_debit AND reverse_count = 0 THEN
        NULL; -- The repository transitions the journal to audited void before adding inverse effects.
    ELSE
        RAISE EXCEPTION 'duplicate or invalid General Ledger effect for journal line %', NEW.journal_line_id
            USING ERRCODE = '23514';
    END IF;

    RETURN NEW;
END;
$function$

guard_order_stock_items|CREATE OR REPLACE FUNCTION public.guard_order_stock_items()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'pg_catalog', 'public'
AS $function$
DECLARE parent_status TEXT; source_row RECORD;
BEGIN
    IF TG_OP = 'DELETE' THEN source_row := OLD; ELSE source_row := NEW; END IF;
    IF TG_OP = 'INSERT' AND NEW.cancelled_quantity <> 0 THEN
        RAISE EXCEPTION 'New items cannot carry unaudited cancellations' USING ERRCODE = '23514';
    END IF;
    IF TG_OP = 'UPDATE' AND (NEW.order_id <> OLD.order_id OR NEW.tenant_id <> OLD.tenant_id OR NEW.restaurant_id <> OLD.restaurant_id) THEN
        RAISE EXCEPTION 'Order item scope is immutable' USING ERRCODE = '23514';
    END IF;
    SELECT status INTO parent_status FROM public.orders
    WHERE id = source_row.order_id AND tenant_id = source_row.tenant_id AND restaurant_id = source_row.restaurant_id FOR UPDATE;
    IF parent_status IS NOT NULL AND parent_status <> 'pending' THEN
        IF TG_OP <> 'UPDATE' OR current_user <> 'pos_migrator' OR parent_status <> 'confirmed'
           OR NEW.cancelled_quantity < OLD.cancelled_quantity
           OR (to_jsonb(NEW) - ARRAY['cancelled_quantity','total_price','updated_at']) <> (to_jsonb(OLD) - ARRAY['cancelled_quantity','total_price','updated_at']) THEN
            RAISE EXCEPTION 'Only audited pre-fulfillment item cancellations are permitted' USING ERRCODE = '23514';
        END IF;
    END IF;
    RETURN source_row;
END;
$function$
